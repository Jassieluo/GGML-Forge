add_executable(nn-model-schema model_schema_cli.cpp)
target_link_libraries(nn-model-schema PRIVATE ggml_nn_ext)
target_include_directories(nn-model-schema PRIVATE ${CMAKE_SOURCE_DIR}/include)

if (INFERENCE_BUILD_TTS)
    target_compile_definitions(nn-model-schema PRIVATE FORGE_SCHEMA_HAS_TTS=1)
    target_link_libraries(nn-model-schema PRIVATE gpt_sovits_models)
    target_include_directories(nn-model-schema PRIVATE ${CMAKE_SOURCE_DIR}/src/categories/tts)
endif()
if (INFERENCE_BUILD_VISUAL_PERCEPTION AND
    (INFERENCE_BUILD_INSTANCE_PERCEPTION OR INFERENCE_BUILD_IMAGE_CLASSIFICATION))
    target_compile_definitions(nn-model-schema PRIVATE FORGE_SCHEMA_HAS_YOLO=1)
    target_link_libraries(nn-model-schema PRIVATE yolo_models)
endif()
if (INFERENCE_BUILD_VISUAL_PERCEPTION AND INFERENCE_BUILD_SEMANTIC_SEGMENTATION)
    target_compile_definitions(nn-model-schema PRIVATE FORGE_SCHEMA_HAS_LRASPP=1)
    target_link_libraries(nn-model-schema PRIVATE lraspp_models)
    target_include_directories(nn-model-schema PRIVATE
        ${CMAKE_SOURCE_DIR}/src/categories/visual_perception/semantic_segmentation)
endif()
if (INFERENCE_BUILD_VISUAL_PERCEPTION AND INFERENCE_BUILD_DEPTH_ESTIMATION)
    target_compile_definitions(nn-model-schema PRIVATE FORGE_SCHEMA_HAS_FASTDEPTH=1)
    target_link_libraries(nn-model-schema PRIVATE fastdepth_models stereonet_models)
    target_include_directories(nn-model-schema PRIVATE
        ${CMAKE_SOURCE_DIR}/src/categories/visual_perception/depth_estimation)
endif()

if (MSVC)
    target_compile_options(nn-model-schema PRIVATE /EHsc /utf-8 /W3)
else()
    target_compile_options(nn-model-schema PRIVATE -Wall -Wextra)
endif()
